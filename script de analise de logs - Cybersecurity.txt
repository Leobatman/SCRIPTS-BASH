#!/usr/bin/env bash
#
# forensic_log_analyzer.sh — Analise forense de logs de acesso web (Blue Team)
# Autor : Leonardo Pereira (Leobatman)
# Versao: 2.0
#
# Uso:
#   ./forensic_log_analyzer.sh <arquivo_de_log> [diretorio_de_saida]
#
# Suporta Common/Combined Log Format (Apache/Nginx), incluindo arquivos .gz.
# Faz a triagem inicial em resposta a incidentes: vetores OWASP, scanners,
# anomalias de trafego, geracao de IOCs e lista de bloqueio de IPs.
#
# Observacao de escopo: ferramenta DEFENSIVA, apenas leitura dos logs.
# Nao executa acoes contra alvos nem modifica o sistema analisado.

set -uo pipefail

# ----------------------------------------------------------------------------
# Argumentos e configuracao
# ----------------------------------------------------------------------------
usage() {
    cat <<EOF
Uso: $0 <arquivo_de_log> [diretorio_de_saida]

  <arquivo_de_log>      Log de acesso (texto ou .gz) no formato Combined.
  [diretorio_de_saida]  Opcional. Padrao: ./forensic_analysis_<timestamp>

Exemplo:
  $0 /var/log/nginx/access.log ./casos/incidente-042
EOF
    exit 2
}

[[ $# -lt 1 || "$1" == "-h" || "$1" == "--help" ]] && usage

readonly LOG_FILE="$1"
readonly TIMESTAMP="$(date +%Y%m%d_%H%M%S)"
readonly OUTPUT_DIR="${2:-./forensic_analysis_${TIMESTAMP}}"
readonly EVIDENCE_DIR="${OUTPUT_DIR}/evidence"
readonly STATS_DIR="${OUTPUT_DIR}/statistics"
readonly INTEL_DIR="${OUTPUT_DIR}/threat_intelligence"
readonly WORK_DIR="${OUTPUT_DIR}/.work"
readonly REPORT_FILE="${OUTPUT_DIR}/security_analysis_report.txt"
readonly MANIFEST="${OUTPUT_DIR}/MANIFEST_sha256.txt"

# Dados normalizados: ip \t requisicao(decodificada) \t status \t user_agent
readonly PARSED="${WORK_DIR}/parsed.tsv"
readonly IPREQ="${WORK_DIR}/ip_request.tsv"

# Limiar adaptativo de "IP suspeito" (volume anormal); ajustado apos a leitura.
IP_THRESHOLD=100

# Cores: usadas apenas no terminal. O relatorio em arquivo fica sempre limpo.
if [[ -t 1 ]]; then
    RED=$'\033[0;91m'; GRN=$'\033[0;92m'; YEL=$'\033[0;93m'
    BLU=$'\033[0;94m'; MAG=$'\033[0;95m'; CYN=$'\033[0;96m'
    BLD=$'\033[1m'; NC=$'\033[0m'
else
    RED=''; GRN=''; YEL=''; BLU=''; MAG=''; CYN=''; BLD=''; NC=''
fi

declare -gA THREAT=([CRITICAL]=0 [HIGH]=0 [MEDIUM]=0 [LOW]=0)
declare -gA VECTORS=([SQLi]=0 [XSS]=0 [LFI]=0 [RCE]=0 [TRAVERSAL]=0 [SCANNING]=0)

# ----------------------------------------------------------------------------
# Saida: terminal colorido + relatorio em texto limpo
# ----------------------------------------------------------------------------
_plain() { printf '%s\n' "$*" >> "$REPORT_FILE"; }

say()      { printf '%s%s%s\n' "$BLU" "$*" "$NC"; _plain "$*"; }
section()  { printf '\n%s%s%s%s\n' "$MAG" "$BLD" "$*" "$NC"
             printf '%s%s%s\n' "$MAG" "==============================================" "$NC"
             _plain ""; _plain "### $*"; }
sub()      { printf '\n%s%s%s%s\n' "$CYN" "$BLD" "$*" "$NC"; _plain ""; _plain "-- $*"; }

critical() { printf '%s%s[CRITICO]%s %s\n' "$RED" "$BLD" "$NC" "$*"; _plain "[CRITICO] $*"; ((THREAT[CRITICAL]++)); }
high()     { printf '%s[ALTO]%s %s\n'      "$RED" "$NC" "$*"; _plain "[ALTO] $*";    ((THREAT[HIGH]++)); }
medium()   { printf '%s[MEDIO]%s %s\n'     "$YEL" "$NC" "$*"; _plain "[MEDIO] $*";   ((THREAT[MEDIUM]++)); }
low()      { printf '%s[BAIXO]%s %s\n'     "$GRN" "$NC" "$*"; _plain "[BAIXO] $*";   ((THREAT[LOW]++)); }
ok()       { printf '%s%s[OK]%s %s\n'      "$GRN" "$BLD" "$NC" "$*"; _plain "[OK] $*"; }

have() { command -v "$1" >/dev/null 2>&1; }

cleanup() { [[ -n "${WORK_DIR:-}" && -d "$WORK_DIR" ]] && rm -rf "$WORK_DIR"; }
trap 'printf "\n%s%sInterrompido. Encerrando...%s\n" "$RED" "$BLD" "$NC"; cleanup; exit 130' INT TERM
trap cleanup EXIT

# ----------------------------------------------------------------------------
# Leitura e normalizacao
# ----------------------------------------------------------------------------
reader() {
    # Le texto puro ou .gz de forma transparente.
    case "$LOG_FILE" in
        *.gz) zcat -- "$LOG_FILE" ;;
        *)    cat  -- "$LOG_FILE" ;;
    esac
}

decode() {
    # URL-decode (%XX e '+') para uma unica passagem. Perl/python se disponiveis;
    # senao, fallback em bash puro.
    if have perl; then
        perl -pe 's/\+/ /g; s/%([0-9a-fA-F]{2})/chr(hex($1))/ge'
    elif have python3; then
        python3 -c 'import sys,urllib.parse as u
for l in sys.stdin: sys.stdout.write(u.unquote_plus(l))'
    else
        local line d
        while IFS= read -r line; do
            d="${line//+/ }"; d="${d//%/\\x}"
            printf '%b\n' "$d"
        done
    fi
}

build_parsed() {
    section "NORMALIZACAO DOS DADOS"
    # Extrai ip / requisicao / status / user-agent assumindo Combined Log Format:
    #   ip - - [data] "METODO /url HTTP/1.1" status bytes "referer" "ua"
    reader | awk -F'"' 'NF>=6 {
        split($1, a, " "); ip = a[1];
        req = $2;
        split($3, s, " "); code = s[1];
        ua = $6;
        gsub(/\t/, " ", req); gsub(/\t/, " ", ua);
        if (ip != "" && req != "")
            print ip "\t" req "\t" code "\t" ua
    }' | decode > "$PARSED"

    cut -f1,2 "$PARSED" > "$IPREQ"

    local total; total=$(wc -l < "$PARSED" | tr -d ' ')
    if [[ "$total" -eq 0 ]]; then
        critical "Nenhuma linha reconhecida no formato esperado (Combined Log Format)."
        exit 1
    fi
    # Limiar adaptativo: 1% do volume total, com piso de 100.
    IP_THRESHOLD=$(( total / 100 )); [[ "$IP_THRESHOLD" -lt 100 ]] && IP_THRESHOLD=100
    ok "Linhas normalizadas: ${total} | Limiar de IP suspeito: ${IP_THRESHOLD}"
}

# ----------------------------------------------------------------------------
# Validacao e setup
# ----------------------------------------------------------------------------
validate_environment() {
    section "VALIDACAO DO AMBIENTE"
    [[ -f "$LOG_FILE" ]] || { critical "Arquivo nao encontrado: $LOG_FILE"; exit 1; }
    [[ -r "$LOG_FILE" ]] || { critical "Sem permissao de leitura: $LOG_FILE"; exit 1; }
    if [[ "$LOG_FILE" == *.gz ]] && ! have zcat; then
        critical "Arquivo .gz informado, mas 'zcat' nao esta disponivel."; exit 1
    fi
    ok "Log valido: $LOG_FILE"
    have perl || have python3 || medium "perl/python3 ausentes: URL-decode usara fallback em bash (mais lento)."
}

setup_dirs() {
    local d
    for d in "$OUTPUT_DIR" "$EVIDENCE_DIR" "$STATS_DIR" "$INTEL_DIR" "$WORK_DIR"; do
        mkdir -p "$d" || { printf 'Falha ao criar %s\n' "$d" >&2; exit 1; }
    done
    {
        echo "RELATORIO DE ANALISE FORENSE"
        echo "Gerado em    : $(date '+%Y-%m-%d %H:%M:%S %z')"
        echo "Log analisado: $LOG_FILE"
        echo "Analisador   : $0 (v2.0)"
        echo "=============================================="
    } > "$REPORT_FILE"
}

# ----------------------------------------------------------------------------
# Custodia: hash do original ANTES de qualquer processamento
# ----------------------------------------------------------------------------
collect_forensics() {
    section "CADEIA DE CUSTODIA"
    local hash size mtime
    hash=$(sha256sum "$LOG_FILE" | cut -d' ' -f1)
    size=$(du -h "$LOG_FILE" | cut -f1)
    mtime=$(stat -c '%y' "$LOG_FILE" 2>/dev/null || stat -f '%Sm' "$LOG_FILE")
    say "SHA-256 do log : $hash"
    say "Tamanho        : $size"
    say "Modificado em  : $mtime"
    {
        echo "# MANIFESTO DE INTEGRIDADE (SHA-256)"
        echo "# Fonte preservada antes do processamento:"
        echo "$hash  $(basename "$LOG_FILE")"
        echo ""
        echo "# Evidencias geradas por esta analise:"
    } > "$MANIFEST"
}

# ----------------------------------------------------------------------------
# Deteccao de vetores de ataque (regex com contexto, sobre a requisicao decodificada)
# ----------------------------------------------------------------------------
# detect <rotulo> <severidade> <chave_vetor> <arquivo_evidencia> <regex_estendida>
detect() {
    local label="$1" sev="$2" key="$3" ev="$4" regex="$5"
    local match="${WORK_DIR}/m_${key}.txt" count ips
    grep -Ei -- "$regex" "$IPREQ" > "$match" 2>/dev/null || true
    count=$(wc -l < "$match" | tr -d ' ')
    if [[ "$count" -gt 0 ]]; then
        cut -f1 "$match" | sort -u > "$EVIDENCE_DIR/$ev"
        ips=$(wc -l < "$EVIDENCE_DIR/$ev" | tr -d ' ')
        VECTORS[$key]=$count
        "$sev" "$label: ${count} requisicoes suspeitas, ${ips} IPs unicos (evidence/$ev)"
    else
        ok "$label: nenhum indicador encontrado"
    fi
}

run_detections() {
    section "DETECCAO DE AMEACAS (OWASP)"

    detect "SQL Injection"  critical SQLi      sqli_ips.txt \
        "union[[:space:]]+select|select[[:space:]].+[[:space:]]from|insert[[:space:]]+into|drop[[:space:]]+table|update[[:space:]].+set|'[[:space:]]*or[[:space:]]*'?[0-9]|[[:space:]]or[[:space:]]+1=1|--[[:space:]-]|/\*.*\*/|information_schema|xp_cmdshell|(sleep|benchmark|pg_sleep)[[:space:]]*\("

    detect "XSS"            high     XSS       xss_ips.txt \
        "<[[:space:]]*script|javascript:|on(error|load|mouseover|click)[[:space:]]*=|<[[:space:]]*img[^>]+src|alert[[:space:]]*\(|String\.fromCharCode|document\.cookie|<[[:space:]]*svg"

    detect "LFI / arquivos sensiveis" high LFI lfi_ips.txt \
        "/etc/passwd|/etc/shadow|/proc/self|php://(filter|input)|file://|expect://|boot\.ini|win\.ini|/windows/win"

    detect "Path Traversal" high     TRAVERSAL traversal_ips.txt \
        "(\.\./){2,}|\.\.[/\\]|\.\.%2f|%2e%2e%2f|\.\.%5c"

    # RCE: exige metachar de shell antes do comando (evita falso positivo com 'id', 'cat' etc.)
    detect "RCE / Command Injection" critical RCE rce_ips.txt \
        "[;|&\`][[:space:]]*(id|whoami|uname|cat|ls|wget|curl|nc|ncat|bash|sh|python|perl)\b|\\\$\([[:space:]]*[a-z]|%3b[[:space:]]*(id|whoami|uname|cat)|/bin/(bash|sh)[[:space:]]*-|system[[:space:]]*\(|exec[[:space:]]*\("
}

# ----------------------------------------------------------------------------
# Analises estatisticas
# ----------------------------------------------------------------------------
analyze_ips() {
    section "ANALISE DE ENDERECOS IP"
    sub "Top 20 IPs por volume"
    cut -f1 "$PARSED" | sort | uniq -c | sort -nr | head -20 \
        | tee -a "$REPORT_FILE" > "$STATS_DIR/top_ips.txt"
    local unique; unique=$(cut -f1 "$PARSED" | sort -u | wc -l | tr -d ' ')
    say "IPs distintos: $unique"

    sub "IPs com volume anormal (>= ${IP_THRESHOLD} req)"
    cut -f1 "$PARSED" | sort | uniq -c | sort -nr \
        | awk -v t="$IP_THRESHOLD" '$1 >= t {print $2"\t"$1" requisicoes"}' \
        > "$EVIDENCE_DIR/high_volume_ips.txt"
    if [[ -s "$EVIDENCE_DIR/high_volume_ips.txt" ]]; then
        high "IPs com volume anormal detectados (evidence/high_volume_ips.txt)"
        tee -a "$REPORT_FILE" < "$EVIDENCE_DIR/high_volume_ips.txt"
    else
        ok "Nenhum IP com volume anormal"
    fi
}

analyze_scanners() {
    section "DETECCAO DE SCANNERS E FERRAMENTAS"
    local agents=(nmap sqlmap nikto metasploit burp acunetix nessus w3af zaproxy gobuster dirb ffuf wpscan masscan hydra)
    local a count total=0
    : > "$EVIDENCE_DIR/scanner_ips.txt"
    for a in "${agents[@]}"; do
        # Procura no user-agent (coluna 4), case-insensitive.
        count=$(awk -F'\t' -v p="$a" 'tolower($4) ~ p {print}' "$PARSED" | wc -l | tr -d ' ')
        if [[ "$count" -gt 0 ]]; then
            high "Scanner '$a': $count requisicoes"
            awk -F'\t' -v p="$a" 'tolower($4) ~ p {print $1}' "$PARSED" | sort -u \
                >> "$EVIDENCE_DIR/scanner_ips.txt"
            total=$((total + count))
        fi
    done
    sort -u -o "$EVIDENCE_DIR/scanner_ips.txt" "$EVIDENCE_DIR/scanner_ips.txt"
    VECTORS[SCANNING]=$total
    [[ "$total" -eq 0 ]] && ok "Nenhum scanner conhecido identificado"
}

analyze_http() {
    section "ANALISE HTTP"
    sub "Metodos"
    awk -F'\t' '{split($2,r," "); print r[1]}' "$PARSED" | sort | uniq -c | sort -nr \
        | tee -a "$REPORT_FILE" > "$STATS_DIR/http_methods.txt"
    local danger
    danger=$(awk -F'\t' '{split($2,r," "); m=r[1]} m ~ /^(PUT|DELETE|TRACE|CONNECT|PATCH)$/ {c++} END{print c+0}' "$PARSED")
    [[ "$danger" -gt 0 ]] && medium "Metodos sensiveis (PUT/DELETE/TRACE/CONNECT): $danger req"

    sub "Codigos de status"
    cut -f3 "$PARSED" | sort | uniq -c | sort -nr \
        | tee -a "$REPORT_FILE" > "$STATS_DIR/status_codes.txt"
    local s5 c4
    c4=$(awk -F'\t' '$3 ~ /^4[0-9][0-9]$/ {c++} END{print c+0}' "$PARSED")
    s5=$(awk -F'\t' '$3 ~ /^5[0-9][0-9]$/ {c++} END{print c+0}' "$PARSED")
    say "Erros 4xx: $c4 | Erros 5xx: $s5"
    [[ "$s5" -gt 100 ]] && high "Volume elevado de erros 5xx ($s5): possivel exploracao ou DoS"

    # Rajada de 401/403 por IP: indicio de brute-force / enumeracao de auth.
    sub "Possivel brute-force de autenticacao (401/403 por IP)"
    awk -F'\t' '$3 ~ /^40[13]$/ {print $1}' "$PARSED" | sort | uniq -c | sort -nr \
        | awk '$1 >= 20 {print $2"\t"$1" falhas"}' > "$EVIDENCE_DIR/auth_bruteforce_ips.txt"
    if [[ -s "$EVIDENCE_DIR/auth_bruteforce_ips.txt" ]]; then
        high "IPs com muitas respostas 401/403 (evidence/auth_bruteforce_ips.txt)"
        tee -a "$REPORT_FILE" < "$EVIDENCE_DIR/auth_bruteforce_ips.txt"
    else
        ok "Nenhum padrao claro de brute-force de autenticacao"
    fi
}

# ----------------------------------------------------------------------------
# Threat intelligence e relatorio
# ----------------------------------------------------------------------------
build_intel() {
    section "INTELIGENCIA DE AMEACAS"
    cat "$EVIDENCE_DIR"/*_ips.txt 2>/dev/null | awk 'NF{print $1}' | sort -u \
        > "$INTEL_DIR/malicious_ips.txt"
    local n; n=$(wc -l < "$INTEL_DIR/malicious_ips.txt" | tr -d ' ')
    if [[ "$n" -gt 0 ]]; then
        high "IPs maliciosos consolidados: $n"
        awk '{print $1" # atividade maliciosa detectada"}' "$INTEL_DIR/malicious_ips.txt" \
            > "$INTEL_DIR/ip_blocklist.txt"
        # Exemplos prontos para aplicar manualmente (NAO aplicados pelo script).
        awk '{print "iptables -A INPUT -s "$1" -j DROP"}' "$INTEL_DIR/malicious_ips.txt" \
            > "$INTEL_DIR/iptables_suggested.sh"
    else
        ok "Nenhum IP malicioso consolidado"
    fi
    {
        echo "# Indicators of Compromise (IOCs)"
        echo "# Gerado em: $(date '+%Y-%m-%d %H:%M:%S %z')"
        echo "IPs maliciosos: $n"
        echo "Vetores:"
        local k
        for k in "${!VECTORS[@]}"; do
            [[ "${VECTORS[$k]}" -gt 0 ]] && echo "  $k: ${VECTORS[$k]}"
        done
    } > "$INTEL_DIR/iocs.txt"
}

executive_summary() {
    section "RESUMO EXECUTIVO"
    {
        echo "Total de linhas    : $(wc -l < "$PARSED" | tr -d ' ')"
        echo "IPs distintos      : $(cut -f1 "$PARSED" | sort -u | wc -l | tr -d ' ')"
        echo ""
        echo "Niveis de ameaca   : CRITICO=${THREAT[CRITICAL]} ALTO=${THREAT[HIGH]} MEDIO=${THREAT[MEDIUM]} BAIXO=${THREAT[LOW]}"
        echo "Vetores detectados :"
    } | tee -a "$REPORT_FILE"
    local k
    for k in "${!VECTORS[@]}"; do
        [[ "${VECTORS[$k]}" -gt 0 ]] && printf '  %-10s %s\n' "$k" "${VECTORS[$k]}" | tee -a "$REPORT_FILE"
    done

    echo "" | tee -a "$REPORT_FILE"
    if   [[ "${THREAT[CRITICAL]}" -gt 0 ]]; then high "Acao imediata: incidentes criticos detectados. Isolar/bloquear e acionar a equipe."
    elif [[ "${THREAT[HIGH]}"     -gt 0 ]]; then high "Atencao: investigar IPs maliciosos e reforcar regras de WAF/firewall."
    elif [[ "${THREAT[MEDIUM]}"   -gt 0 ]]; then medium "Monitorar: atividade suspeita, sem incidente critico."
    else ok "Sem ameacas significativas no periodo analisado."
    fi
}

finalize() {
    # Hash de todas as evidencias geradas (integridade pos-analise).
    find "$EVIDENCE_DIR" "$STATS_DIR" "$INTEL_DIR" -type f -print0 2>/dev/null \
        | sort -z | xargs -0 sha256sum 2>/dev/null >> "$MANIFEST"

    local archive="${OUTPUT_DIR}.tar.gz"
    tar -czf "$archive" -C "$(dirname "$OUTPUT_DIR")" "$(basename "$OUTPUT_DIR")" 2>/dev/null \
        && ok "Pacote forense: $archive"
    ok "Resultados em: $OUTPUT_DIR"
}

# ----------------------------------------------------------------------------
main() {
    printf '%s%s\n== ANALISADOR FORENSE DE LOGS v2.0 ==%s\n' "$BLU" "$BLD" "$NC"
    validate_environment
    setup_dirs
    collect_forensics
    build_parsed
    analyze_ips
    analyze_http
    analyze_scanners
    run_detections
    build_intel
    executive_summary
    finalize
    printf '%s%sAnalise concluida.%s\n' "$GRN" "$BLD" "$NC"
}

main "$@"
